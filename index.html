function playBeep(freq, dur, reps) { try { if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)(); if (audioCtx.state === 'suspended') audioCtx.resume(); if (phase === 'WORK') flash(reps === 1 ? 'flash-o' : 'flash-y'); else if (phase === 'PREP') flash('flash-b'); for (let i = 0; i < reps; i++) { setTimeout(() => { const osc = audioCtx.createOscillator(); const gain = audioCtx.createGain(); osc.connect(gain); gain.connect(audioCtx.destination); osc.frequency.value = freq; // VOLUME INCREASED HERE (from 0.1 to 0.5) gain.gain.setValueAtTime(0.5, audioCtx.currentTime); // This creates a quick fade out to prevent "popping" sounds gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + dur); osc.start(); osc.stop(audioCtx.currentTime + dur); }, i * 200); } } catch(e) { console.error(e); } }
